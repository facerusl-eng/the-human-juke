 130: function acquireIntroAudioPlayLock(eventId: string): string | null {
 131:   if (typeof window === 'undefined') {
 132:     return null
 133:   }
 134: 
 135:   const now = Date.now()
 136:   const ownerId = `${eventId}:${now}:${Math.random().toString(36).slice(2)}`
 137: 
 138:   try {
 139:     const existingRaw = window.localStorage.getItem(INTRO_AUDIO_LOCK_STORAGE_KEY)
 140: 
 141:     if (existingRaw) {
 142:       const existingLock = JSON.parse(existingRaw) as Partial<IntroAudioPlayLock>
 143:       const sameEvent = existingLock.eventId === eventId
 144:       const stillValid = typeof existingLock.expiresAt === 'number' && existingLock.expiresAt > now
 145: 
 146:       if (sameEvent && stillValid) {
 147:         return null
 148:       }
 149:     }
 150: 
 151:     const nextLock: IntroAudioPlayLock = {
 152:       eventId,
 153:       ownerId,
 154:       expiresAt: now + INTRO_AUDIO_LOCK_TTL_MS,
 155:     }
 156: 
 157:     window.localStorage.setItem(INTRO_AUDIO_LOCK_STORAGE_KEY, JSON.stringify(nextLock))
 158: 
 159:     const confirmationRaw = window.localStorage.getItem(INTRO_AUDIO_LOCK_STORAGE_KEY)
 160:     if (!confirmationRaw) {
 161:       return null
 162:     }
 163: 
 164:     const confirmation = JSON.parse(confirmationRaw) as Partial<IntroAudioPlayLock>
 165:     return confirmation.ownerId === ownerId && confirmation.eventId === eventId ? ownerId : null
 166:   } catch {
 167:     return ownerId
 168:   }

 270: async function fetchServerClockOffsetMs(): Promise<number | null> {
 271:   if (typeof window === 'undefined') {
 272:     return null
 273:   }
 274: 
 275:   const requestStartedAt = Date.now()
 276: 
 277:   try {
 278:     const response = await fetch(`/api/keepwarm?clock-sync=${Date.now()}`, {
 279:       method: 'GET',
 280:       cache: 'no-store',
 281:     })
 282: 
 283:     if (!response.ok) {
 284:       return null
 285:     }
 286: 
 287:     const requestEndedAt = Date.now()
 288:     const serverDateHeader = response.headers.get('date')
 289: 
 290:     if (!serverDateHeader) {
 291:       return null
 292:     }
 293: 
 294:     const serverNowMs = Date.parse(serverDateHeader)
 295: 
 296:     if (!Number.isFinite(serverNowMs)) {
 297:       return null
 298:     }
 299: 
 300:     const estimatedClientNowMs = Math.round((requestStartedAt + requestEndedAt) / 2)
 301:     return serverNowMs - estimatedClientNowMs
 302:   } catch {
 303:     return null
 304:   }
